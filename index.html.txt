<script>
// 自動載入記住的 API Key
window.onload = function() {
  const savedKey = localStorage.getItem('gemini_key');
  if (savedKey) {
    document.getElementById('apiKeyInput').value = savedKey;
  }
};

async function callGeminiAndExport() {
  const apiKey = document.getElementById('apiKeyInput').value.trim();

  if (!apiKey) {
    alert("請先在最上方的綠色框框貼上你的 Google Gemini API Key 喔！");
    document.getElementById('apiKeyInput').focus();
    return;
  }

  // 儲存金鑰以便下次使用
  localStorage.setItem('gemini_key', apiKey);

  const school = document.getElementById('school').value;
  const teacher = document.getElementById('teacher').value;
  const theme = document.getElementById('theme').value;
  const activity = document.getElementById('activity').value;
  const target = document.getElementById('target').value;
  const duration = document.getElementById('duration').value;
  const domain = document.getElementById('domain').value;
  const highlight = document.getElementById('highlight').value;

  const submitBtn = document.getElementById('submitBtn');
  const loadingText = document.getElementById('loadingText');

  submitBtn.disabled = true;
  loadingText.style.display = "block";

  const prompt = `你現在是一位資深的幼兒園教保專業人員，熟悉臺灣《幼兒園教保活動課程大綱》（課綱六大領域：身體動作與健康、認知、語文、社會、情緒、美感）。
請針對以下條件，為我設計一份完整、現場可直接執行的主題活動教案：
【基本資訊】
園所名稱：${school}
設計教師：${teacher}
主題名稱：${theme}
活動名稱：${activity}
教學對象：${target}
活動時間：${duration}
主要核心領域：${domain}
特殊構想：${highlight || "無特別指定，請根據年齡設計適齡操作"}

【格式強制要求】
請不要輸出 Markdown 標題符號，請直接輸出純 HTML 程式碼片段（不要包在 \`\`\`html 區塊內），直接從 <div> 開始，結構包含：
1. 一、活動設計理念（約 60 字，扣合幼兒生活經驗）
2. 二、教學目標與課綱指標對應（以 <table> 呈現，包含目標與指標代碼如：認-中-1-1-1）
3. 三、活動準備（教師器材、幼兒自備、場地佈置）
4. 四、活動流程（以 <table> 呈現，分為引起動機、發展活動步驟1-3與具體師生引導語、綜合活動與收拾）
5. 五、學習評量指標（以 <table> 呈現，分為優良、通過、需引導三級）
6. 六、教學省思與延伸學習（安全與班控、學習區延伸規劃、特殊需求幼兒鷹架）
所有表格請加上 border="1" style="border-collapse:collapse; width:100%;"`;

  try {
    const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        contents: [{ parts: [{ text: prompt }] }]
      })
    });

    const data = await response.json();
    if (!response.ok) {
      throw new Error(data.error ? data.error.message : "API 呼叫失敗，請檢查 API Key 是否正確。");
    }

    let generatedHtml = data.candidates[0].content.parts[0].text;
    generatedHtml = generatedHtml.replace(/```html/g, '').replace(/```/g, '');

    const fullWordContent = `
    <html xmlns:o='urn:schemas-microsoft-com:office:office' xmlns:w='urn:schemas-microsoft-com:office:word' xmlns='http://www.w3.org/TR/REC-html40'>
    <head>
      <meta charset='utf-8'>
      <title>${theme} - ${activity} 教案</title>
      <style>
        body { font-family: '標楷體', 'PMingLiU', sans-serif; font-size: 12pt; line-height: 1.5; }
        h1 { text-align: center; font-size: 18pt; margin-bottom: 10px; font-weight: bold; }
        table { width: 100%; border-collapse: collapse; margin-top: 10px; margin-bottom: 15px; }
        th, td { border: 1px solid #000; padding: 6px 8px; text-align: left; vertical-align: top; }
        th { background-color: #f2f2f2; font-weight: bold; }
        .header-info { margin-bottom: 15px; }
      </style>
    </head>
    <body>
      <h1>【幼兒園主題活動教案】</h1>
      <div class="header-info">
        <p><b>園所名稱：</b>${school}  <b>設計教師：</b>${teacher}</p>
        <p><b>活動主題：</b>${theme}  <b>活動名稱：</b>${activity}  <b>教學對象：</b>${target}</p>
      </div>
      <hr/>
      ${generatedHtml}
    </body>
    </html>
    `;

    const blob = new Blob(['\ufeff' + fullWordContent], { type: 'application/msword' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${school}_${theme}_${activity}_教案.doc`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);

  } catch (err) {
    alert("生成失敗，原因：" + err.message);
  } finally {
    submitBtn.disabled = false;
    loadingText.style.display = "none";
  }
}
</script>
